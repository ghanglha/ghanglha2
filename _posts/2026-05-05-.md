བུད་མེད་ཀྱི་སེམས་ཁྲལ་དང་སེམས་ངལ་གྱི་ནད་གཞི་དེ་འཛམ་གླིང་ཡོངས་སུ་སེམས་ཁམས་འཕྲོད་བསྟེན་གྱི་དཀའ་ངལ་ཆེ་ཤོས་ཤིག་ཡིན།

 🧠 སེམས་ཁྲལ་དང་སེམས་ངལ་ཟེར་བ་གང་ཡིན་ནམ།

 སེམས་ཁྲལ་གྱི་ནད་གཞི་དེ་ཉིན་རེའི་འཚོ་བར་བར་ཆད་བཟོ་སྲིད་པའི་སེམས་ཁྲལ་ཚད་ལས་བརྒལ་བ་དང་། འཇིགས་སྣང་། ཡང་ན་སེམས་ཁྲལ་བཅས་ཚུད་ཡོད།

 སེམས་ངལ་གྱི་ནད་གཞི་ནི་སེམས་སྡུག་དང་རེ་བ་མེད་པ་དང་བྱ་འགུལ་ལ་སྤྲོ་བ་ཤོར་བའི་ཚོར་སྣང་རྒྱུན་འཁྱོངས་བྱེད་པ་དེ་རེད།

 བུད་མེད་ལ་གནས་སྟངས་འདི་དག་འབྱུང་བའི་གོ་སྐབས་ཕོ་སྐྱེས་ལས་ལྡབ་གཉིས་ཙམ་ཡོད།

 🌸 ཅིའི་ཕྱིར་བུད་མེད་ལ་ཤུགས་རྐྱེན་མང་བ་ཐེབས་ཡོད་དམ།

 1\. གཤེར་ཁུའི་ཤུགས་རྐྱེན།

 ཧོར་མོན་གྱི་འགྱུར་བ་དེས་འགན་གཙོ་བོ་འཁུར་གྱི་ཡོད།

 ལང་ཚོ་དར་བབས

 ཟླ་མཚན་གྱི་འཁོར་ཡུག 

 མངལ་ཆགས་པ།

 སྐྱེས་རྗེས་ཀྱི་དུས་ཡུན།

 ཟླ་མཚན་འགག་པ།

 འགྱུར་ལྡོག་འདི་དག་གིས་སེམས་ཁམས་དང་ཀླད་པའི་རྫས་སྦྱོར་ལ་ཐད་ཀར་ཤུགས་རྐྱེན་ཐེབས་སྲིད།

 2\. མི་ཚེའི་ཉམས་མྱོང་དང་སྤྱི་ཚོགས་ཀྱི་གནོན་ཤུགས།

 བུད་མེད་ཚོས་རྒྱུན་དུ་འཕྲད་པ་ནི།

 ཕོ་མོའི་དབྱེ་འབྱེད་ཀྱི་དྲག་གནོན་ཚད་མཐོ་བ།

 ལྟ་སྐྱོང་འགན་འཁྲི།

 ལས་ཀ་དང་འཚོ་བའི་དོ་མཉམ་མེད་པ།

 ཕྱི་ཚུལ་དང་འགན་འཁྲིའི་སྐོར་གྱི་སྤྱི་ཚོགས་ཀྱི་རེ་བ།

 གནོན་ཤུགས་འདི་དག་གིས་སེམས་ཁམས་འཕྲོད་བསྟེན་གྱི་གནད་དོན་ལ་རྨས་སྐྱོན་ཕོག་ཚད་ཆེ་རུ་གཏོང་ཐུབ།

 3\. སྐྱེ་ལྡན་གནོན་ཤུགས་ལ་ཚོར་ཤུགས།

 ཞིབ་འཇུག་གིས་བུད་མེད་ཚོས་སེམས་ངལ་ལ་ཐབས་ལམ་མི་འདྲ་བ་སྤྱོད་ཐུབ་པ་སྟོན་གྱི་ཡོད།![](/assets/images/IMG_2845.jpeg)

 ⚠️ རྒྱུན་མཐོང་གི་ནད་རྟགས།

 སེམས་ཁྲལ་གྱི་རྟགས།

 རྒྱུན་དུ་སེམས་ཁྲལ་བྱེད་པའམ་ཡང་ན་བསམ་བློ་ཚད་ལས་བརྒལ་བ།

 བདེ་འཇགས་མེད་པའམ་ཡང་ན་“མཐའ་མཚམས་སུ་”ཡོད་པའི་ཚོར་སྣང་།

 གཉིད་དཀའ་བ།

 སྙིང་གི་འཕར་ལྡིང་མགྱོགས་པ།

 སེམས་ངལ་གྱི་ནད་རྟགས།

 རྒྱུན་མི་ཆད་པར་སྐྱོ་བའམ་སྟོང་པ།

 དགའ་ཕྱོགས་ལ་སྤྲོ་སྣང་ཤོར་བ།

 ཐང་ཆད་པའམ་ནུས་ཤུགས་ཞན་པ།

 ཟས་ཀྱི་འདོད་པའམ་གཉིད་ཀྱི་འགྱུར་བ།

 ཉེས་སྐྱོན་དང་རིན་ཐང་མེད་པའི་ཚོར་བ།

 👶 དམིགས་བསལ་གནད་དོན།སྐྱེས་རྗེས་ཀྱི་སེམས་ཁམས་འཕྲོད་བསྟེན།

 ཕྲུ་གུ་བཙས་རྗེས་བུད་མེད་ཁ་ཤས་ལ་མྱོང་ཚོར་བྱུང་བ་ནི།

 “མ་འབྱར་གྱི་སེམས་ཁྲལ་”(དུས་ཡུན་ཐུང་ཐུང་གི་སེམས་ཁམས་འཕོ་འགྱུར།)

 སྐྱེས་རྗེས་ཀྱི་སེམས་ངལ་(ཚབས་ཆེ་བ་དང་། ཡུན་རིང་གནས་པ)

 གནས་སྟངས་འདི་ཚབས་ཆེན་ཡིན་པས་རྒྱབ་སྐྱོར་དང་ལྟ་སྐྱོང་ཡག་པོ་བྱེད་དགོས།

 🛠️ སྨན་བཅོས་དང་རྒྱབ་སྐྱོར།

 1\. ཆེད་ལས་རོགས་རམ།

 སྨན་བཅོས། (དཔེར་ན། འཛིན་སྟངས་སྤྱོད་ལམ་གྱི་སྨན་བཅོས།)

 དགོས་དུས་སྨན་བསྟེན་དགོས།

 2\. འཚོ་བའི་ཐབས་ཤེས།

 རྒྱུན་དུ་ལུས་སྦྱོང་བྱེད་པ།

 འཚོ་བཅུད་དོ་མཉམ།

 གཉིད་ཀྱི་གོམས་གཤིས་ཡག་པོ།

 སེམས་ཁམས་རིག་པའམ་སྒོམ་རྒྱག་པ།

 3\. སྤྱི་ཚོགས་ཀྱི་རྒྱབ་སྐྱོར།

 གྲོགས་པོའམ་ནང་མི་དང་ཁ་བརྡ་བྱེད་སྐབས།

 རྒྱབ་སྐྱོར་ཚོགས་པ།

 སྤྱི་ཚོགས་སམ་དྲ་རྒྱའི་ཐོན་ཁུངས།

 💡 ཅིའི་ཕྱིར་གོ་རྟོགས་གལ་ཆེན་པོ་རེད།

 བུད་མེད་མང་པོ་ཞིག་གིས་མིང་ཆད་དང་ཤེས་རྟོགས་མེད་པའི་རྐྱེན་གྱིས་རོགས་རམ་འཚོལ་གྱི་མེད།  ཡིན་ནའང་སེམས་ཁྲལ་དང་སེམས་ངལ་གྱི་ནད་དེ་སྨན་བཅོས་བྱེད་ཐུབ་པ་དང་། སྔ་མོ་ནས་རྒྱབ་སྐྱོར་བྱས་ན་ཁྱད་པར་ཆེན་པོ་བཟོ་ཐུབ།

 གལ་ཏེ་ཁྱེད་ལ་དགོས་ན། ངས་འདི་ཁྱེད་ལ་དགོས་མཁོ་ལྟར་གསུང་བཤད་དང་། རྩོམ་ཡིག ཡང་ན་གསལ་བཤད་ཀྱི་བརྙན་ཤེལ་ལ་བསྒྱུར་ཐུབ།
